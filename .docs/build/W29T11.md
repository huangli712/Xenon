## 实施步骤

### Step 1: 创建测试文件与导入

创建或更新目标文件，使用 `xenon::prelude::*` 和 `tests/common` helper。每个测试名称必须描述可观察行为，并保持单文件可独立运行。

```rust
use xenon::prelude::*;

#[test]
fn test_unique_set_equality() {
    // 28-tests §5.12 L524: unique returns distinct elements; compared to input
    // by multiset semantics; NaN elements compared by occurrence count.
    let t = Tensor1::from_shape_vec(
        [6],
        vec![1.0_f64, f64::NAN, 2.0, 1.0, f64::NAN, 2.0],
    )
    .unwrap();
    // 实证 14-set.md L181：`pub fn unique(&self) -> Tensor<A, Ix1>` 返 Tensor 非 Result，无需 .unwrap()。
    let u = t.unique();

    let input_nan_count = 2usize;
    let uniq_nan_count = u.iter().filter(|v| v.is_nan()).count();
    assert_eq!(
        uniq_nan_count, input_nan_count,
        "NaN elements must be compared by occurrence count",
    );

    let mut uniq_non_nan: Vec<f64> =
        u.iter().copied().filter(|v| !v.is_nan()).collect();
    uniq_non_nan.sort_by(|a, b| a.partial_cmp(b).unwrap());
    assert_eq!(
        uniq_non_nan,
        vec![1.0, 2.0],
        "distinct non-NaN elements must match input multiset",
    );
}

#[test]
fn test_unique_integers() {
    let t = Tensor1::from_shape_vec([6], vec![1, 2, 2, 3, 1, 3]).unwrap();
    let u = t.unique();  // 返 Tensor 非 Result（8个位置同样修复）
    let mut vals: Vec<i32> = u.iter().copied().collect();
    vals.sort_unstable();
    assert_eq!(vals, vec![1, 2, 3]);
}

#[test]
fn test_unique_complex() {
    let a = Complex::new(1.0, 2.0);
    let b = Complex::new(3.0, 4.0);
    let t = Tensor1::from_shape_vec([3], vec![a, b, a]).unwrap();
    let u = t.unique();
    assert_eq!(u.len(), 2);
}

#[test]
fn test_unique_nan_preserved() {
    // 28-tests §5.12 L527: `NaN != NaN`, every NaN in the input must be preserved.
    let t = Tensor1::from_shape_vec([4], vec![1.0_f64, f64::NAN, 2.0, f64::NAN]).unwrap();
    let u = t.unique();
    let nan_count = u.iter().filter(|v| v.is_nan()).count();
    assert_eq!(nan_count, 2, "every NaN in input must be preserved");
    let mut non_nan: Vec<f64> = u.iter().copied().filter(|v| !v.is_nan()).collect();
    non_nan.sort_by(|a, b| a.partial_cmp(b).unwrap());
    assert_eq!(non_nan, vec![1.0, 2.0]);
}
```

### Step 2: 实现正常路径、边界路径和错误路径

按 28-tests §5 的清单补齐正常路径、边界输入、错误返回或编译失败 fixture；涉及浮点、复数、SIMD、并行时按 §6.2 和 §6.3 选择对应比较层级。

```rust
#[test]
fn test_unique_non_contiguous() {
    // 命名严格对齐 28-tests §5.12 L530（无 _view 后缀）。
    // 14-set §8.2: unique on a non-contiguous view operates on logical elements only.
    // Transposing a 2x3 tensor yields a 3x2 view whose strides no longer match
    // F-order, so the underlying buffer traversal is non-contiguous.
    let owner = Tensor2::<f64>::from_shape_vec(
        [2, 3],
        vec![1.0, 2.0, 1.0, 3.0, 2.0, 3.0],
    )
    .expect("shape and data length must match");

    let view = owner.transpose();
    let uniq = view.unique();

    assert_eq!(uniq.len(), 3);
}

#[test]
fn test_unique_transposed_view() {
    // 额外覆盖（非 §5.12 表内）：transpose 是非连续的常见特例，
    // 14-set §8.2 明确"unique 应按逻辑元素去重，不受 layout 影响"。
    // 本测试用 i32 元素验证数值正确性，与 test_unique_non_contiguous（f64, 仅长度）形成互补。
    let owner = Tensor2::<i32>::from_shape_vec(
        [2, 3],
        vec![1, 4, 2, 5, 3, 6],
    )
    .expect("shape and data length must match");

    let t = owner.transpose();
    let uniq = t.unique();

    let mut vals: Vec<i32> = uniq.iter().copied().collect();
    vals.sort_unstable();
    assert_eq!(vals, vec![1, 2, 3, 4, 5, 6]);
}
```

```rust
#[test]
fn test_unique_signed_zero_equal() {
    // 28-tests §5.12 L528: `-0.0` and `0.0` are treated as equal;
    // only one zero value is preserved.
    let t = Tensor1::from_shape_vec([4], vec![1.0_f64, 0.0, -0.0, 2.0]).unwrap();
    let u = t.unique();
    let zero_count = u.iter().filter(|v| **v == 0.0).count();
    assert_eq!(zero_count, 1, "+0.0 and -0.0 must collapse to a single zero");
}

#[test]
fn test_unique_order_unspecified() {
    // 28-tests §5.12 L525: lint anchor — tests must not depend on any
    // particular output order. Compare as a set, never as an ordered vector.
    let t = Tensor1::from_shape_vec([3], vec![3, 1, 2]).unwrap();
    let u = t.unique();  // 返 Tensor 非 Result，无需 .unwrap()
    let mut vals: Vec<i32> = u.iter().copied().collect();
    vals.sort_unstable();
    assert_eq!(vals, vec![1, 2, 3]);
}
```
