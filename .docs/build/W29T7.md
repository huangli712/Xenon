## 实施步骤

### Step 1: 创建测试文件与导入

创建或更新目标文件，使用 `xenon::prelude::*` 和 `tests/common` helper。每个测试名称必须描述可观察行为，并保持单文件可独立运行。

```rust
use xenon::prelude::*;
use xenon::error::XenonError;

#[test]
fn test_zeros_ones_from_scalar() {
    // W22T2/T3/T8 设计：zeros / ones / from_scalar 均返回 Result<Self, XenonError>
    // （28-tests §L833 均一致使用 .expect("shape is valid")）。
    let z = Tensor1::<f64>::zeros([3]).expect("shape is valid");
    assert_eq!(z.shape(), &[3]);
    assert!(z.iter().all(|v| *v == 0.0));

    let o = Tensor2::<f64>::ones([2, 2]).expect("shape is valid");
    assert_eq!(o.shape(), &[2, 2]);
    assert!(o.iter().all(|v| *v == 1.0));

    let s = Tensor0::from_scalar(42.0_f64).expect("scalar must construct");
    assert_eq!(s.shape(), &[] as &[usize]);
    assert_eq!(*s.iter().next().unwrap(), 42.0);
}

#[test]
fn test_eye_identity() {
    let e = Tensor2::<f64>::eye(3).expect("eye(3) is valid");
    assert_eq!(e.shape(), &[3, 3]);
    assert_eq!(*e.try_at(&[0, 0]).unwrap(), 1.0);
    assert_eq!(*e.try_at(&[0, 1]).unwrap(), 0.0);
    assert_eq!(*e.try_at(&[1, 1]).unwrap(), 1.0);
    assert_eq!(*e.try_at(&[2, 2]).unwrap(), 1.0);
    assert_eq!(*e.try_at(&[2, 0]).unwrap(), 0.0);
}

#[test]
fn test_from_data_constructors() {
    // from_shape_vec: owned vector path.
    let t1 = Tensor1::from_shape_vec([3], vec![1.0_f64, 2.0, 3.0]).unwrap();
    assert_eq!(t1.shape(), &[3]);
    assert_eq!(*t1.try_at(&[0]).unwrap(), 1.0);

    // from_shape_slice: borrowed slice path (§7 T7).
    let src = [10.0_f64, 20.0, 30.0, 40.0];
    let t2 = Tensor2::from_shape_slice([2, 2], &src).unwrap();
    assert_eq!(t2.shape(), &[2, 2]);
    assert_eq!(*t2.try_at(&[0, 0]).unwrap(), 10.0);
    assert_eq!(*t2.try_at(&[1, 1]).unwrap(), 40.0);

    // Length mismatch must return XenonError::InvalidShape, not a bare is_err().
    let err = Tensor1::<f64>::from_shape_vec([5], vec![1.0, 2.0]).unwrap_err();
    assert!(matches!(err, XenonError::InvalidShape { .. }));
}
```

### Step 2: 实现正常路径、边界路径和错误路径

按 28-tests §5.8 清单补齐 `test_from_fixed_array`（从固定数组构造）与 `test_from_shape_vec_f_order_mapping`（F-order 逻辑元素顺序与线性输入映射的高优先级覆盖）。错误路径已在 Step 1 中以 `XenonError::InvalidShape` 精确断言。

```rust
#[test]
fn test_from_fixed_array() {
    // 18-construction.md §5.3 L424：from_array<Sh, const N>(shape: Sh, arr: [A; N])
    // ——两参数签名，shape 默认获得 N 不够。
    let arr: [f64; 2] = [10.0, 20.0];
    let t = Tensor1::from_array([2], arr).expect("shape matches array length");
    assert_eq!(t.shape(), &[2]);
    assert_eq!(*t.try_at(&[0]).unwrap(), 10.0);
    assert_eq!(*t.try_at(&[1]).unwrap(), 20.0);
}

#[test]
fn test_from_shape_vec_f_order_mapping() {
    // Logical element order is F-order (column-major): for shape [2, 3],
    // the linear input [v0..v5] maps to logical coordinates as
    // v0=(0,0), v1=(1,0), v2=(0,1), v3=(1,1), v4=(0,2), v5=(1,2).
    let data = vec![10.0_f64, 11.0, 12.0, 13.0, 14.0, 15.0];
    let t = Tensor2::from_shape_vec([2, 3], data).unwrap();
    assert_eq!(*t.try_at(&[0, 0]).unwrap(), 10.0);
    assert_eq!(*t.try_at(&[1, 0]).unwrap(), 11.0);
    assert_eq!(*t.try_at(&[0, 1]).unwrap(), 12.0);
    assert_eq!(*t.try_at(&[1, 1]).unwrap(), 13.0);
    assert_eq!(*t.try_at(&[0, 2]).unwrap(), 14.0);
    assert_eq!(*t.try_at(&[1, 2]).unwrap(), 15.0);
}
```

| 决策 | 依据 |
