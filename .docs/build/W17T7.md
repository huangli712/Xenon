use xenon::{dot, Complex, Ix1, Ix2, Tensor, Tensor1, XenonError};

/// 12-matrix §10.1 line 574-575: f64 dot tolerance.
fn f64_dot_tolerance(n: usize, max_abs_a: f64, max_abs_b: f64) -> f64 {
    let ulp_term = 8.0 * f64::EPSILON * (n as f64) * max_abs_a * max_abs_b;
    let floor = 4.0 * f64::MIN_POSITIVE;
    ulp_term.max(floor)
}

/// 12-matrix §10.1 line 573: f32 dot tolerance.
#[allow(dead_code)]
fn f32_dot_tolerance(n: usize, max_abs_a: f32, max_abs_b: f32) -> f32 {
    let ulp_term = 8.0 * f32::EPSILON * (n as f32) * max_abs_a * max_abs_b;
    let floor = 4.0 * f32::MIN_POSITIVE;
    ulp_term.max(floor)
}

### Step 7: 跨路径浮点容差 — `test_dot_float_tolerance_across_paths`

§8.2 line 476 (高优先级)。此前缺失，本 task 在集成层落地。测试构造确定性输入，用 §10.1 公式与"手算 sum-of-products" 基线对比，无论 dispatch 选哪条路径（Serial / Simd / Parallel）都必须落在容差内。

```rust
#[test]
fn test_dot_float_tolerance_across_paths() {
    // §10.1 line 574-575 + §8.6 line 511-512: regardless of feature
    // configuration (default / simd / parallel / simd+parallel), the
    // observable result must fall within the documented tolerance
    // relative to a single-precision-fold scalar baseline.
    let n: usize = 8192;
    let a_vals: Vec<f64> = (0..n).map(|i| ((i % 17) as f64) * 0.1 + 1.0).collect();
    let b_vals: Vec<f64> = (0..n).map(|i| ((i % 19) as f64) * 0.07 + 0.5).collect();
    let a = Tensor1::from_shape_vec(Ix1(n), a_vals.clone()).unwrap();
    let b = Tensor1::from_shape_vec(Ix1(n), b_vals.clone()).unwrap();

    // Reference computed at integration layer via a single-pass
    // sequential fold — the only baseline the public API guarantees
    // tolerance against.
    let expected: f64 = a_vals
        .iter()
        .zip(b_vals.iter())
        .fold(0.0_f64, |acc, (x, y)| acc + x * y);

    let actual = dot(&a, &b).unwrap();

    let max_abs_a = a_vals.iter().fold(0.0_f64, |acc, &v| acc.max(v.abs()));
    let max_abs_b = b_vals.iter().fold(0.0_f64, |acc, &v| acc.max(v.abs()));
    let tol = f64_dot_tolerance(n, max_abs_a, max_abs_b);

    let diff = (actual - expected).abs();
    assert!(
        diff <= tol,
        "dot diverges from scalar baseline: actual={actual}, expected={expected}, diff={diff}, tol={tol}"
    );
}
```

### Step 8: feature-gate 一致性 — `test_dot_simd_parallel_combined_consistency` / `test_dot_parallel_threshold_boundary`

§8.2 line 472 (双特性组合) + line 473 (并行阈值边界)。两者都用容差比较 (不是 `is_finite()`)。

```rust
#[cfg(all(feature = "simd", feature = "parallel"))]
#[test]
fn test_dot_simd_parallel_combined_consistency() {
    // §8.2 line 472: SIMD + parallel combined path matches scalar
    // baseline within §10.1 tolerance.
    let n: usize = 16_384;
    let a_vals: Vec<f64> = (0..n).map(|i| (i as f64) * 0.001 + 1.0).collect();
    let b_vals: Vec<f64> = (0..n).map(|i| (i as f64) * 0.001 + 2.0).collect();
    let a = Tensor1::from_shape_vec(Ix1(n), a_vals.clone()).unwrap();
    let b = Tensor1::from_shape_vec(Ix1(n), b_vals.clone()).unwrap();

    let actual = dot(&a, &b).unwrap();
    let expected: f64 = a_vals
        .iter()
        .zip(b_vals.iter())
        .fold(0.0_f64, |acc, (x, y)| acc + x * y);

    let max_abs_a = a_vals.iter().fold(0.0_f64, |acc, &v| acc.max(v.abs()));
    let max_abs_b = b_vals.iter().fold(0.0_f64, |acc, &v| acc.max(v.abs()));
    let tol = f64_dot_tolerance(n, max_abs_a, max_abs_b);

    assert!((actual - expected).abs() <= tol);
}

#[cfg(feature = "parallel")]
#[test]
fn test_dot_parallel_threshold_boundary() {
    // §8.2 line 473 + §8.3 line 486: threshold-adjacent inputs (just
    // below, equal-to, and just above) maintain correct path selection
    // and result semantics. Run three deterministic vector sizes; each
    // must match the scalar baseline within §10.1 tolerance.
    //
    // feature-gated on "parallel" because the test's value comes from
    // verifying dispatch's serial<->parallel transition; without the
    // parallel feature, all three sizes go through Serial unconditionally
    // and the test degrades to a triple scalar consistency check.
    for &n in &[4095_usize, 4096, 4097] {
        let a_vals: Vec<f64> = (0..n).map(|i| ((i % 7) as f64) + 1.0).collect();
        let b_vals: Vec<f64> = (0..n).map(|i| ((i % 5) as f64) + 1.0).collect();
        let a = Tensor1::from_shape_vec(Ix1(n), a_vals.clone()).unwrap();
        let b = Tensor1::from_shape_vec(Ix1(n), b_vals.clone()).unwrap();

        let actual = dot(&a, &b).unwrap();
        let expected: f64 = a_vals
            .iter()
            .zip(b_vals.iter())
            .fold(0.0_f64, |acc, (x, y)| acc + x * y);

        let max_abs_a = a_vals.iter().fold(0.0_f64, |acc, &v| acc.max(v.abs()));
        let max_abs_b = b_vals.iter().fold(0.0_f64, |acc, &v| acc.max(v.abs()));
        let tol = f64_dot_tolerance(n, max_abs_a, max_abs_b);

        assert!(
            (actual - expected).abs() <= tol,
            "threshold-boundary n={n}: actual={actual}, expected={expected}, tol={tol}"
        );
    }
}
```

## 关键设计决策

- **§8.2 测试函数命名严格对齐设计清单**：所有测试函数名一字不差复用 §8.2 line 459-476 列出的名称，未自创 `_consistency` 等后缀（设计列表中显式有 `_consistency` 的 `test_dot_simd_parallel_combined_consistency` 保留其原名）。
- **跨路径比较全部用 §10.1 容差**：`is_finite()` 退化测试被全部替换为"actual 与确定性 sum-of-products 基线之差 ≤ §10.1 公式" 的容差断言；浮点 SIMD / Parallel 在 ISA / chunk 顺序变化下不再 flaky。
- **集成层不重复 unit-level pub(crate) 测试**：`test_dot_simd_path_with_feature`（admission gate 行为）已在 W17T5 单元测试落地；`test_dot_parallel_path` / `test_dot_large_vector_parallel_threshold` / `test_dot_nested_parallel_falls_back`（需要 `dispatch::with_parallel_worker_context`）在 W17T6 单元测试落地。设计 §8.2 line 459-476 共 16 条高优先级测试，本文件在集成层覆盖剩下 12 条（含 `test_dot_float_tolerance_across_paths`），与 §8.5 line 503 "端到端协同验证" 的集成测试边界一致。
- **`xenon::dot` 通过 crate root 直接可见**：W17T1 已落地 `pub use matrix::dot;`，本文件 `use xenon::{dot, ...};` 一行闭环。
- **method-style 与 free-function 在 `test_dot_basic` 双覆盖**：避免单独开 method 测试导致拆分，与 §5.1 line 171 "method 必须与 free function 共享相同的契约" 一致。
