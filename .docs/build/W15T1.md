```rust
//! See `docs/design/09-parallel.md` for the full design.

pub(crate) mod reduce;
pub(crate) mod checked;

// Module-level foundational utility (09-parallel §6.3 line 338).
// Defined here rather than in a separate file to match design specification.
/// Compute a per-chunk element count for parallel splitting.
///
/// Algorithm (09-parallel §6.3):
/// - `MIN_CHUNK = 1024`           : lower bound to amortize rayon scheduling.
/// - `TARGET_CHUNKS_PER_WORKER=4` : give work-stealing slack for tail balance.
/// - `total == 0`        → returns 1 (dummy; no work will be scheduled).
/// - `total <= workers`  → returns 1 (one element per worker; rest idle).
/// - otherwise           → max(ceil_div(total, workers*4), MIN_CHUNK).
pub(crate) fn compute_safe_chunks(total: usize, num_workers: usize) -> usize {
    const MIN_CHUNK: usize = 1024;
    const TARGET_CHUNKS_PER_WORKER: usize = 4;

    if total == 0 {
        return 1;
    }
    if total <= num_workers {
        return 1;
    }
    let target_chunks = num_workers.saturating_mul(TARGET_CHUNKS_PER_WORKER);
    let raw = total.div_ceil(target_chunks);
    raw.max(MIN_CHUNK)
}
```

> **职责边界**：本 step 仅声明子模块名并实现 `compute_safe_chunks`。`map` / `reduce` / `checked` 三个子文件由 W15T2-T7 各自创建并填充内容。`mod.rs` 本身在 W15T6 由 `ParallelPool` 增量扩展。

### Step 3: 在 `src/parallel/iter.rs` 定义 `ParIter` 与 `with_strategy()`

`ParIter` 是 `pub(crate)`，持有 `TensorView` + 当前逻辑区间 `[lo, hi)` + 来自 `ParallelExecStrategy` 的两个字段（设计 §5.6 line 252-261）。

```rust
// src/parallel/iter.rs
use crate::dispatch::ParallelExecStrategy;
use crate::dimension::Dimension;
use crate::element::Element;
use crate::tensor::{TensorBase, TensorView};
use crate::storage::Storage;

#[cfg(feature = "parallel")]
pub(crate) struct ParIter<'a, A, D>
where
    A: Element + Send + Sync,
    D: Dimension,
{
    base: TensorView<'a, A, D>,
    /// Logical index interval [lo, hi) currently owned by this producer.
    /// Initially [0, base.len()); rayon splits sub-ranges via Producer::split_at.
    lo: usize,
    hi: usize,
    chunk_size: Option<usize>,
    max_workers: Option<usize>,
}

#[cfg(feature = "parallel")]
impl<'a, A, D> ParIter<'a, A, D>
where
    A: Element + Send + Sync,
    D: Dimension,
{
    /// Construct with default strategy fields. Used by `TensorBase::par_iter()`.
    pub(crate) fn new(base: TensorView<'a, A, D>) -> Self {
        let hi = base.len();
        Self { base, lo: 0, hi, chunk_size: None, max_workers: None }
    }

    /// Override strategy fields after construction (used by par_map_checked etc.).
    /// See 09-parallel §5.6 line 272.
    pub(crate) fn with_strategy(mut self, strategy: &ParallelExecStrategy) -> Self {
        self.chunk_size = strategy.chunk_size();
        self.max_workers = strategy.max_workers();
        self
    }
}

// `Clone` is required by `par_map_checked` (W15T7) to run the two-pass
// pattern without cloning the user closure: phase 1 consumes a cloned
// ParIter via `try_for_each`, phase 2 consumes the original. The clone
// is metadata-only (TensorView + 4 small scalar fields) — no element
// data is duplicated. A manual impl is used (not `#[derive(Clone)]`)
// so the `A: Clone` bound that derive would emit is not propagated;
// `D: Dimension` already implies `Clone` per 02-dimension §5.1 line 123,
// and `TensorView: Clone` is provided by `tensor` (07-tensor §5).
#[cfg(feature = "parallel")]
impl<'a, A, D> Clone for ParIter<'a, A, D>
where
    A: Element + Send + Sync,
    D: Dimension,
{
    fn clone(&self) -> Self {
        Self {
            base: self.base.clone(),
            lo: self.lo,
            hi: self.hi,
            chunk_size: self.chunk_size,
            max_workers: self.max_workers,
        }
    }
}
```

### Step 4: 实现 `Producer` 与 `IndexedParallelIterator`

> **设计决策注释**：`ParIter` struct 包含 `lo`/`hi` 字段以支持 `Producer::split_at` 的 disjoint coverage 不变量（09-parallel §5.6 line 265-269）。`split_at` 实现通过逻辑索引区间拆分而非指针算术，推迟到 `ParIterSeq::next()` 中通过切片索引实现 F-contiguous 快速路径——功能等价于设计 §5.6 line 266 的指针算术要求，但实现路径不同。

按 09-parallel §5.6 line 263-272 的拆分契约实现，**F-contiguous 走指针算术、非连续走逻辑区间 stride 状态机**。**重要：30-dispatch §5.6 表明确表示非连续输入不被拒绝进入 Parallel，只是阈值翻倍（`len >= 2 * PARALLEL_THRESHOLD`）；§6.1 算法中 `is_contiguous == false` 只会让 `effective_parallel = base.saturating_mul(2)`，仍会返回 `ExecPath::Parallel`。因此本 task **不能**依赖"dispatch 快速路径只发 F-contiguous"这个错误前提。**本 task 采取 entry-side gating 策略：`TensorBase::par_iter()` 在返回 `ParIter` 前填裝 `debug_assert!(self.is_f_contiguous())`，release 构建下如果 dispatch 错误路由了非 F-contiguous 输入，`ParIterSeq::next()` 的 `unimplemented!()` 会以明确诊断信息 panic。调用侧 (`par_map` / `par_map_checked`) 需在调用 `par_iter()` 前自行检查连续性，非连续输入走 fallback。正确的非连续 stride 状态机推迟到后续 wave（需 17-indexing §5 line 251 `get_unchecked(&[usize])` 与 dimension stride 计算助手全部就绪）。

```rust
#[cfg(feature = "parallel")]
impl<'a, A, D> rayon::iter::ParallelIterator for ParIter<'a, A, D>
where
    A: Element + Send + Sync,
    D: Dimension,
{
    type Item = &'a A;
    fn drive_unindexed<C>(self, consumer: C) -> C::Result
    where
        C: rayon::iter::plumbing::UnindexedConsumer<Self::Item>,
    {
        rayon::iter::plumbing::bridge(self, consumer)
    }
    fn opt_len(&self) -> Option<usize> {
        Some(self.hi - self.lo)
    }
}

#[cfg(feature = "parallel")]
impl<'a, A, D> rayon::iter::IndexedParallelIterator for ParIter<'a, A, D>
where
    A: Element + Send + Sync,
    D: Dimension,
{
    fn len(&self) -> usize {
        self.hi - self.lo
    }

    fn drive<C>(self, consumer: C) -> C::Result
    where
        C: rayon::iter::plumbing::Consumer<Self::Item>,
    {
        rayon::iter::plumbing::bridge(self, consumer)
    }

    fn with_producer<CB>(self, callback: CB) -> CB::Output
    where
        CB: rayon::iter::plumbing::ProducerCallback<Self::Item>,
    {
        callback.callback(self)
    }
}

#[cfg(feature = "parallel")]
impl<'a, A, D> rayon::iter::plumbing::Producer for ParIter<'a, A, D>
where
    A: Element + Send + Sync,
    D: Dimension,
{
    type Item = &'a A;
    type IntoIter = ParIterSeq<'a, A, D>;

    fn into_iter(self) -> Self::IntoIter {
        ParIterSeq::new(self.base, self.lo, self.hi)
    }

    fn split_at(self, index: usize) -> (Self, Self) {
        // Disjoint coverage invariant (09-parallel §5.6 line 268):
        // produces two sub-producers covering [lo, mid) and [mid, hi) with
        // no overlap and no gap.
        debug_assert!(index <= self.hi - self.lo);
        let mid = self.lo + index;
        let left = Self {
            base: self.base.clone(),
            lo: self.lo,
            hi: mid,
            chunk_size: self.chunk_size,
            max_workers: self.max_workers,
        };
        let right = Self {
            base: self.base,
            lo: mid,
            hi: self.hi,
            chunk_size: self.chunk_size,
            max_workers: self.max_workers,
        };
        (left, right)
    }
}

/// Sequential per-chunk iterator. F-contiguous: returns a slice iterator
/// over the contiguous sub-range; non-contiguous: walks logical indices
/// via stride state machine (deferred — see note below).
#[cfg(feature = "parallel")]
pub(crate) struct ParIterSeq<'a, A, D>
where
    A: Element + Send + Sync,
    D: Dimension,
{
    base: TensorView<'a, A, D>,
    cursor: usize,
    end: usize,
}

#[cfg(feature = "parallel")]
impl<'a, A, D> ParIterSeq<'a, A, D>
where
    A: Element + Send + Sync,
    D: Dimension,
{
    fn new(base: TensorView<'a, A, D>, lo: usize, hi: usize) -> Self {
        Self { base, cursor: lo, end: hi }
    }
}

#[cfg(feature = "parallel")]
impl<'a, A, D> Iterator for ParIterSeq<'a, A, D>
where
    A: Element + Send + Sync,
    D: Dimension,
{
    type Item = &'a A;
    fn next(&mut self) -> Option<Self::Item> {
        if self.cursor >= self.end {
            return None;
        }
        let i = self.cursor;
        self.cursor += 1;
        if self.base.is_f_contiguous() {
            // F-contiguous fast path: index directly into the underlying slice.
            Some(&self.base.as_slice().unwrap()[i])
        } else {
            // Non-contiguous walk: deferred to a later wave (see Step 5
            // `par_iter()` entry-side gating + module note above).
            // If reached, it indicates a caller bypassed `par_iter()`'s
            // F-contiguous precondition. The diagnostic must be loud.
            unimplemented!(
                "non-F-contiguous ParIter walk is unimplemented in W15. \
                 Callers MUST gate non-F-contiguous inputs to a serial \
                 fallback before calling par_iter() (see W15T1 Step 5 \
                 entry-side gating). 30-dispatch §5.6 admits non-contig \
                 inputs to Parallel via doubled threshold; this is a real \
                 path that callers must handle, not a dispatch bug."
            )
        }
    }
    fn size_hint(&self) -> (usize, Option<usize>) {
        let remaining = self.end - self.cursor;
        (remaining, Some(remaining))
    }
}

#[cfg(feature = "parallel")]
impl<'a, A, D> ExactSizeIterator for ParIterSeq<'a, A, D>
where
    A: Element + Send + Sync,
    D: Dimension,
{}

#[cfg(feature = "parallel")]
impl<'a, A, D> DoubleEndedIterator for ParIterSeq<'a, A, D>
where
    A: Element + Send + Sync,
    D: Dimension,
{
    fn next_back(&mut self) -> Option<Self::Item> {
        if self.cursor >= self.end {
            return None;
        }
        self.end -= 1;
        let i = self.end;
        if self.base.is_f_contiguous() {
            Some(&self.base.as_slice().unwrap()[i])
        } else {
            unimplemented!(
                "non-F-contiguous ParIter walk is unimplemented in W15. \
                 Callers MUST gate non-F-contiguous inputs to a serial \
                 fallback before calling par_iter()."
            )
        }
    }
}
```

### Step 5: 实现 `TensorBase::par_iter()`

按 09-parallel §5.6 line 282-286 在 `TensorBase` 上添加 `par_iter()`，返回默认策略的 `ParIter`。

```rust
#[cfg(feature = "parallel")]
impl<S, D, A> TensorBase<S, D>
where
    S: Storage<Elem = A>,
    D: Dimension + Clone,
    A: Element + Send + Sync,
{
    /// Single-input element-level parallel traversal entry.
    /// Returns a `ParIter` with default strategy (None/None);
    /// call `.with_strategy()` to override.
    ///
    /// # Entry-side gating (W15 scope)
    ///
    /// In W15, only F-contiguous inputs are supported by the internal
    /// `ParIterSeq::next()`. Non-F-contiguous inputs would trigger an
    /// `unimplemented!()` panic. Although 30-dispatch §5.6 admits
    /// non-contiguous inputs to Parallel via a doubled threshold,
    /// callers MUST self-gate non-F-contiguous inputs to a serial
    /// fallback before invoking `par_iter()`.
    ///
    /// `debug_assert!` documents the contract under debug builds;
    /// release builds still rely on the explicit caller-side check.
    /// A future wave will replace the `unimplemented!()` branch with a
    /// real stride state machine and remove this gating.
    pub(crate) fn par_iter(&self) -> ParIter<'_, A, D> {
        debug_assert!(
            self.is_f_contiguous(),
            "par_iter() in W15 supports F-contiguous inputs only; \
             callers must gate non-F-contiguous inputs to serial. \
             See W15T1 entry-side gating note."
        );
        ParIter::new(self.view())
    }
}
```

`ParIter` only yields `&'a A`, never `&mut A`; mutable parallel access is outside W15 scope (09-parallel §5.6 line 263, §1.1 line 18).

## 关键设计决策

| 决策 | 依据 |
|------|------|
| W15T1 承担模块骨架 + `compute_safe_chunks`（不只 iter.rs） | 09-parallel §3 文件结构 + §6.3 line 338 明确要求函数定义于 `mod.rs`；W15T2-T5 已声明依赖 `compute_safe_chunks`，把它放到 W15T1 消除循环；W1T3 lib.rs 模块声明演进协议 |
| `ParIter` 内部新增 `lo`/`hi` 字段以支持 Producer::split_at | 09-parallel §5.6 line 265-269 producer 拆分契约的字面要求（disjoint coverage）；设计仅列 `base`/`chunk_size`/`max_workers` 三字段，实现扩展为 5 字段以满足 rayon Producer 协议 |
| `split_at` 通过逻辑索引区间拆分，非指针算术 | 09-parallel §5.6 line 266 要求 F-contiguous 用 `ptr.add`；实现推迟到 `ParIterSeq::next()` 中通过切片索引实现——功能等价，实现路径不同 |
| `with_strategy()` 由 W15T1 提供，不延迟到 W15T2 | 09-parallel §5.6 line 272 显式要求；W15T2/W15T4 已立即调用此方法 |
| 非 F-contiguous 路径用 `unimplemented!()` 占位，但 `par_iter()` 入口加 `debug_assert!(is_f_contiguous())` | 30-dispatch §5.6 明确表示非连续走 Parallel 路径（仅阈值翻倍），非"dispatch routes such inputs to Serial"；W15 范围内仅支持 F-contiguous 的 stride walk，调用侧 (par_map/par_map_checked) 负责在 `par_iter()` 前依据 `is_f_contiguous()` 自行走 fallback；debug 断言提供开发阶段可见性 |
| `ParIter` 只读、不提供 `ParIterMut` | 09-parallel §1.1 line 18 + §5.6 line 263 显式范围外 |
| `ParIter` 手写 `Clone` impl（不用 derive），仅复制元数据 | W15T7 `par_map_checked` 两遍模式需要克隆迭代器（设计 §6.6 line 482）；手写 impl 避免 derive 引入冗余 `A: Clone` bound（`A` 是元素类型，按 §5.6 line 252 定义不要求 `Clone`）；`TensorView: Clone` 已由 07-tensor §5 提供，`D: Dimension` 已隐含 `Clone`（02-dimension §5.1 line 123） |
