/// Helper for formatting optional values in error messages.
///
/// Displays `<any>` if `None`, otherwise formats the value via `Display`.
/// The `<any>` text is mandated by `26-error §5.7`.
struct OrAny<T>(Option<T>);

impl<T: fmt::Display> fmt::Display for OrAny<T> {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        match &self.0 {
            Some(v) => write!(f, "{}", v),
            None => write!(f, "<any>"),
        }
    }
}

/// Helper for formatting `[usize]` shape/stride slices in error messages.
///
/// Output format: `[]`、`[5]`、`[2 × 3 × 4]` —— 与 NumPy 风格一致；
/// 见 `26-error §5.7` 的 Display 输出要求。
struct FmtShape<'a>(&'a [usize]);

impl<'a> fmt::Display for FmtShape<'a> {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        write!(f, "[")?;
        for (i, dim) in self.0.iter().enumerate() {
            if i > 0 {
                write!(f, " × ")?;
            }
            write!(f, "{}", dim)?;
        }
        write!(f, "]")
    }
}

/// Display impls for auxiliary enums used in `XenonError::Display`.
/// Each impl produces lowercase, human-readable descriptions.
/// Display impls cover all current variants exhaustively, mirroring the
/// reference impls in `26-error.md §5.1`. When new variants are added in
/// future minor versions, the corresponding Display impl MUST be updated
/// (the compiler's exhaustiveness check catches the omission within this
/// crate, since `#[non_exhaustive]` only enforces wildcard arms in
/// downstream crates).
impl fmt::Display for StorageKindTag {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        match self {
            Self::Owned => write!(f, "owned"),
            Self::View => write!(f, "view"),
            Self::ViewMut => write!(f, "view mut"),
            Self::Shared => write!(f, "shared"),
        }
    }
}

impl fmt::Display for StorageConversionKind {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        match self {
            Self::ToOwned => write!(f, "to owned"),
            Self::IntoOwned => write!(f, "into owned"),
            Self::Transpose => write!(f, "transpose"),
            Self::SliceMut => write!(f, "slice mut"),
            Self::BroadcastTo => write!(f, "broadcast to"),
        }
    }
}

impl fmt::Display for FfiBackend {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        match self {
            Self::RawParts => write!(f, "raw parts"),
            Self::Blas => write!(f, "BLAS"),
        }
    }
}

impl fmt::Display for InvalidLayoutReason {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        match self {
            Self::ShapeProductOverflow => write!(f, "shape product overflow"),
            Self::StridesRankMismatch => write!(f, "strides rank mismatch"),
            Self::AccessRangeExceedsStorage => write!(f, "access range exceeds storage"),
            Self::EmptyTensorOffsetExceedsStorage => write!(f, "empty tensor offset exceeds storage"),
            Self::UnsupportedStride => write!(f, "unsupported stride"),
            Self::StrideExceedsIsizeMax => write!(f, "stride exceeds isize::MAX"),
            Self::StrideSpanOverflow => write!(f, "stride span overflow"),
            Self::AccessRangeOverflow => write!(f, "access range overflow"),
            Self::UnexpectedZeroStride => write!(f, "unexpected zero stride"),
            Self::AmbiguousOverlap => write!(f, "ambiguous overlap"),
            Self::OwnedRequiresZeroOffset => write!(f, "owned requires zero offset"),
            Self::LenShapeMismatch => write!(f, "len-shape mismatch"),
            Self::CapacityBelowLen => write!(f, "capacity below len"),
            Self::AlignmentInvalid => write!(f, "alignment invalid"),

impl fmt::Display for InvalidShapeKind {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        match self {
            Self::ProductOverflow => write!(f, "product overflow"),
            Self::ElementCountMismatch { expected, actual } => {
                write!(f, "element count mismatch: expected {expected}, got {actual}")
            }
            Self::RankExceedsStaticMax { provided_ndim, max_ndim } => {

                write!(f, "invalid config `{argument}`: {constraint}, got {actual}")
            }
            Self::DuplicateOrEmpty { argument } => {
                write!(f, "duplicate or empty `{argument}`")

            Self::OperationSpecific { argument, constraint } => {
                write!(f, "`{argument}`: {constraint}")
            }
        }
    }

impl fmt::Display for FfiErrorCategory {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        match self {
            Self::NullPointer { argument } =>
                write!(f, "null pointer for argument {argument}"),

                write!(f, "alignment mismatch: required {required}, actual {actual}"),

impl fmt::Display for XenonError {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        match self {
            Self::ShapeMismatch { operation, left_shape, right_shape } => {
                write!(
                    f,
                    "shape mismatch in `{operation}`: cannot operate on {} and {}",
                    FmtShape(left_shape),
                    FmtShape(right_shape),
                )
            }
            Self::BroadcastError { operation, lhs_shape, rhs_shape, attempted_target_shape, axis } => {
                write!(
                    f,
                    "broadcast error in `{operation}`: cannot broadcast {} with {}",
                    FmtShape(lhs_shape),
                    FmtShape(rhs_shape),
                )?;
                if let Some(ref target) = attempted_target_shape {
                    write!(f, " (attempted target: {})", FmtShape(target))?;
                }
                if let Some(ax) = axis {
                    write!(f, " (axis: {ax})")?;
                }
                Ok(())
            }
            Self::LayoutMismatch { operation, required_layout, actual_layout, shape } => {
                write!(
                    f,
                    "layout mismatch in `{operation}`: required {required_layout}, got {actual_layout} for shape {}",
                    FmtShape(shape),
                )
            }
            Self::InvalidLayout { operation, storage_kind, shape, strides, offset, storage_len, reason } => {
                write!(
                    f,
                    "invalid layout ({reason}) in `{operation}`: storage={storage_kind}, shape={}, strides={}, offset={offset}, len={storage_len}",
                    FmtShape(shape),
                    FmtShape(strides),
                )
            }
            Self::InvalidAxis { operation, axis, ndim, shape } => {
                write!(
                    f,
                    "invalid axis {axis} in `{operation}`: valid range is 0..{ndim} for shape {}",
                    FmtShape(shape),
                )
            }
            Self::InvalidShape { operation, shape, kind, offending_dim } => {
                write!(
                    f,
                    "invalid shape ({kind}) in `{operation}`: shape={}",
                    FmtShape(shape),
                )?;
                if let Some(dim) = offending_dim {
                    write!(f, " (offending dim: {dim})")?;
                }
                Ok(())
            }
            Self::DimensionMismatch { operation, expected, actual } => {
                write!(
                    f,
                    "dimension mismatch in `{operation}`: expected {expected} dimensions, got {actual}",
                )
            }
            Self::InvalidArgument { operation, kind } => {
                write!(
                    f,
                    "invalid argument ({kind}) in `{operation}`",
                )
            }
            Self::InvalidStorageMode { operation, expected, actual, shape, conversion } => {
                write!(
                    f,
                    "invalid storage mode in `{operation}`: expected {expected}, got {actual}",
                )?;
                if let Some(ref s) = shape {
                    write!(f, " for shape {}", FmtShape(s))?;
                }
                if let Some(c) = conversion {
                    write!(f, " (conversion: {c})")?;
                }
                Ok(())
            }
            Self::Ffi { operation, category, backend, cause } => {
                write!(
                    f,
                    "FFI error (`{category}`) in `{operation}` (backend: {backend})",
                )?;
                if let Some(ref inner) = cause {
                    // Contract from `26-error §5.7`: append `; caused by: <inner>`
                    // so a single Display call shows the whole chain. Programmatic
                    // traversal still uses `std::error::Error::source()`.
                    write!(f, "; caused by: {inner}")?;
                }
                Ok(())
            }
            Self::Workspace { operation, category, cause } => {
                write!(
                    f,
                    "workspace error (`{category}`) in `{operation}`",
                )?;
                if let Some(ref inner) = cause {
                    // Same chain contract as `Ffi` above; see `26-error §5.7`.
                    write!(f, "; caused by: {inner}")?;
                }
                Ok(())
            }
            Self::IndexOutOfBounds { operation, attempted_index, axis, shape } => {
                write!(
                    f,
                    "index out of bounds in `{operation}`: attempted {} at axis {axis} (shape: {})",
                    FmtShape(attempted_index),
                    FmtShape(shape),
                )
            }
            Self::TypeConversion { operation, source_type, target_type, reason, element_index } => {
                write!(
                    f,
                    "type conversion failed in `{operation}`: {source_type} -> {target_type} ({reason})",
                )?;
                if let Some(idx) = element_index {
                    write!(f, " at element index {idx}")?;
                }
                Ok(())
            }
        }
    }
}

## 关键设计决策

| 决策 | 依据 |
|------|------|
| `FmtShape` 用 `[a × b × c]` 格式 | `26-error §5.7` Display 契约 + NumPy 风格 |
| `OrAny<T>::None` 渲染为 `<any>` | `26-error §5.7` 明确规定 `None` 统一显示为 `<any>` |
| 辅助枚举实现 `Display` | 避免 XenonError::Display 中混用 `{:?}`（Debug）和 `{}`（Display），所有辅助类型（`InvalidLayoutReason`、`StorageKindTag`、`FfiErrorCategory` 等）提供可读的 `Display` 输出，fallback 到 Debug 以兼容未来 `#[non_exhaustive]` 变体 |
| `cause: Some(_)` 时追加 `; caused by: <inner>` | `26-error §5.7` 公共契约；程序化遍历仍通过 `Error::source()` |
| `Option` 字段为 `None` 时条件省略 | 保持无详情时的错误信息简洁，且不输出 Debug 的 `Some/None` 字面量 |

## 验证方式

2. **测试**: `cargo test -- error`
3. **手动**: 构造各变体并 println! 输出，确认信息可读

## 注意事项

- Display 输出面向用户，不应包含内部实现细节
- `FmtShape` 在 0-dim 输出 `[]`（空维度标记）
- 条件 `details` 字段的输出需要适当换行控制
