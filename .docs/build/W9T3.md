## 实施步骤

### Step 1: 实现核心代码

在 `src/workspace/workspace.rs` 中添加本任务要求的最小实现，保持职责边界与设计文档一致。

```rust
pub struct Workspace {
    pub(crate) ptr: core::ptr::NonNull<u8>,
    pub(crate) capacity: usize,
    pub(crate) alignment: usize,
    pub(crate) borrow_state: core::sync::atomic::AtomicU8,
    pub(crate) split_count: core::sync::atomic::AtomicUsize,
    _not_send_sync: core::marker::PhantomData<*mut ()>,
}

impl Workspace {
    pub const DEFAULT_ALIGNMENT: usize = 64;
    pub const MIN_ALIGNMENT: usize = 8;
    pub const DEFAULT_CAPACITY: usize = 4096;

    const BORROW_NONE: u8 = 0;
    const BORROW_READ: u8 = 1;
    const BORROW_EXCLUSIVE: u8 = 2;

    const GROWTH_FACTOR_NUMERATOR: usize = 3;
    const GROWTH_FACTOR_DENOMINATOR: usize = 2;

    pub fn capacity(&self) -> usize { self.capacity }
    pub fn alignment(&self) -> usize { self.alignment }

    pub fn new(capacity: usize, alignment: usize) -> crate::Result<Self> {
        if !alignment.is_power_of_two() || alignment < Self::MIN_ALIGNMENT {
            return Err(crate::error::XenonError::Workspace {
                operation: alloc::borrow::Cow::Borrowed("Workspace::new"),
                category: crate::error::WorkspaceErrorCategory::InvalidLayout {
                    size: capacity,
                    align: alignment,
                },
                cause: None,
            });
        }
        let size = capacity.max(1);
        let layout = alloc::alloc::Layout::from_size_align(size, alignment)
            .map_err(|_| crate::error::XenonError::Workspace {
                operation: alloc::borrow::Cow::Borrowed("Workspace::new"),
                category: crate::error::WorkspaceErrorCategory::InvalidLayout {
                    size: capacity,
                    align: alignment,
                },
                cause: None,
            })?;
        let ptr = if size == 0 {
            core::ptr::NonNull::dangling()
        } else {
            // SAFETY: layout is non-zero size and valid alignment
            let raw = unsafe { alloc::alloc::alloc(layout) };
            core::ptr::NonNull::new(raw).ok_or_else(|| {
                crate::error::XenonError::Workspace {
                    operation: alloc::borrow::Cow::Borrowed("Workspace::new"),
                    category: crate::error::WorkspaceErrorCategory::AllocFailed {
                        size,
                        align: alignment,
                    },
                    cause: None,
                }
            })?
        };
        Ok(Self {
            ptr,
            capacity: size,
            alignment,
            borrow_state: core::sync::atomic::AtomicU8::new(Self::BORROW_NONE),
            split_count: core::sync::atomic::AtomicUsize::new(0),
            _not_send_sync: core::marker::PhantomData,
        })
    }

    pub fn with_default_capacity() -> crate::Result<Self> {
        Self::new(Self::DEFAULT_CAPACITY, Self::DEFAULT_ALIGNMENT)
    }
}

// Drop 与设计文档 §5.4 一致：`new()` 保证 `self.capacity = capacity.max(1) >= 1`，
// `self.ptr` 总是有效分配的指针（`alloc::alloc::alloc(layout)` 返回的需 dealloc），
// 所以 Drop 可以无条件调用 dealloc，不需额外守护。
impl Drop for Workspace {
    fn drop(&mut self) {
        // SAFETY: layout was valid at allocation time, and ptr is the same.
        unsafe {
            let layout = alloc::alloc::Layout::from_size_align_unchecked(
                self.capacity,
                self.alignment,
            );
            alloc::alloc::dealloc(self.ptr.as_ptr(), layout);
        }
    }
}
```

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn test_workspace_new_default() {
        let ws = Workspace::with_default_capacity().unwrap();
        assert_eq!(ws.capacity(), Workspace::DEFAULT_CAPACITY);
        assert_eq!(ws.alignment(), Workspace::DEFAULT_ALIGNMENT);
    }

    #[test]
    fn test_workspace_new() {
        let ws = Workspace::new(1024, 64).unwrap();
        assert_eq!(ws.capacity(), 1024);
        assert_eq!(ws.alignment(), 64);
    }

    #[test]
    fn test_workspace_constants() {
        assert_eq!(Workspace::DEFAULT_ALIGNMENT, 64);
        assert_eq!(Workspace::MIN_ALIGNMENT, 8);
        assert_eq!(Workspace::DEFAULT_CAPACITY, 4096);
    }

    #[test]
    fn test_workspace_new_invalid_alignment() {
        // alignment 非 2 的幂 → InvalidLayout
        let err = Workspace::new(1024, 7).unwrap_err();
        match err {
            crate::error::XenonError::Workspace {
                category: crate::error::WorkspaceErrorCategory::InvalidLayout { size: 1024, align: 7 },
                ..
            } => {}
            other => panic!("expected InvalidLayout, got {:?}", other),
        }

        // alignment 小于 MIN_ALIGNMENT → InvalidLayout
        let err = Workspace::new(1024, 4).unwrap_err();
        match err {
            crate::error::XenonError::Workspace {
                category: crate::error::WorkspaceErrorCategory::InvalidLayout { size: 1024, align: 4 },
                ..
            } => {}
            other => panic!("expected InvalidLayout, got {:?}", other),
        }
    }

    #[test]
    fn test_workspace_drop_no_leak() {
        let ws = Workspace::new(1024, 64).unwrap();
        drop(ws);
    }
}

## 关键设计决策

| 决策 | 依据 |
|------|------|
| `_not_send_sync: PhantomData<*mut ()>` | `24-workspace.md §5.1` 保持非 Send/Sync |
| `self.capacity = size`（= `capacity.max(1)`） | `24-workspace.md §5.3` L277：保证 `self.capacity >= 1`，避免 `new(0, _)` 时分配了字节但 capacity 记为 0 导致的泄露。`capacity()` 返回请求后的 max(1) 结果 |
| `pub(crate)` 暴露字段给子模块 | borrow/split/expand 子模块需访问内部状态 |
