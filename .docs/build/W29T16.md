#[test]
fn test_broadcast_shape_error() {
    // 逐元素不可广播：shape 冲突走 BroadcastError
    let a = Tensor1::<f64>::from_shape_vec([3], vec![1.0; 3]).unwrap();
    let b = Tensor1::<f64>::from_shape_vec([5], vec![2.0; 5]).unwrap();
    let err = (&a + &b).unwrap_err();
    assert!(matches!(err, XenonError::BroadcastError { .. }),
        "expected BroadcastError, got {err:?}");
    let msg = format!("{err}");
    assert!(msg.contains("broadcast") || msg.contains("shape"),
        "display must mention broadcast/shape: {msg}");
}

#[test]
fn test_invalid_argument_error() {
    // 28-tests §5.23 L695 要求测试 XenonError::InvalidArgument。
    // design/17-indexing.md §5.3.2 L214: SliceInfo::new(indices, in_dim, out_dim)
    // 在 indices.len() != in_dim 等结构性约束失败时返回 InvalidArgument。
    // 这里通过 in_dim/out_dim 与 indices 长度不一致触发该路径。
    // 17-indexing.md §5.3.2 L214：SliceInfo::new(indices, in_dim, out_dim) 三参数。
    // 仅传入 1 个 SliceInfoElem 而 in_dim = Ix2 (rank=2)，indices.len() != in_dim → InvalidArgument。
    // 实证 17-indexing.md L155-161 + L198：第一参数是 `SliceInfoIndices` 枚举
    // （Inline/Dynamic），不是 array；SliceInfo<I, D> 两个泛型参数（输出/输入维度），
    // 不是三个。手工构造 indices.len()=1 + in_dim=Ix2(rank=2)，结构性约束失败 → InvalidArgument。
    let err = SliceInfo::<Ix2, Ix2>::new(
        SliceInfoIndices::Inline {
            len: 1,
            elems: [Some(SliceInfoElem::Range { start: 0, end: 1 }), None, None, None, None, None],
        },
        Ix2(2, 3),  // in_dim：传非零合法形状以避免 zero-dim 验证早返回 InvalidShape（17-indexing.md §5.3.2）。
        Ix2(2, 3),  // out_dim 同上；indices.len()=1 与 in_dim.ndim()=2 不匹配，触发 InvalidArgument。
    )
    .unwrap_err();
    assert!(
        matches!(err, XenonError::InvalidArgument { .. }),
        "expected InvalidArgument, got {err:?}"
    );
    let msg = format!("{err}");
    assert!(
        !msg.is_empty(),
        "Display must be non-empty for InvalidArgument: {msg}"
    );
}

#[test]
fn test_invalid_storage_mode_conversion_error() {
    // 不允许的存储模式转换返回 InvalidStorageMode，且 Display 含转换上下文
    let t = Tensor1::<f64>::from_shape_vec([3], vec![1.0; 3]).unwrap();
    // 实证 05-storage.md L622：ArcTensor 转换公开入口是 `Owned::into_shared`。
    // 在 TensorBase 层体现为 `t.into_shared() -> ArcTensor<A, D>`。
    let arc: ArcTensor1<f64> = t.into_shared();
    // 实证 21-type.md L327：`into_owned(self) -> Tensor<A, D>` 在多重 Arc 引用时
    // 走 deep_clone 路径（§5.4），不返回错误而是物理复制。本测试转为验证
    // ArcTensor.into_owned() 在多重共享时**不丢失数据**且**保持独占语义**——这是
    // 26-error.md 中 InvalidStorageMode 不会被此路径触发的语义证据。
    //
    // ⚠️ 真正能触发 InvalidStorageMode 的路径目前未在公共 API 中暴露（28-tests.md
    // §5.23 L693 / 26-error.md L175 InvalidStorageMode 变体覆盖的是低层 storage
    // conversion 拒绝路径）。本测试做变体存在性 + Display 健康检查，待 W7/W25
    // 提供具体触发入口后再扩展。
    let aliased = arc.clone();
    let owned = arc.into_owned();
    drop(aliased);
    // 验证 deep_clone 路径产出独占 Owned 与原数据等价。
    assert_eq!(owned.iter().copied().collect::<Vec<_>>(), vec![1.0_f64, 1.0, 1.0]);

    // InvalidStorageMode 变体存在性 + Display 健康检查（参照 W29T16 line 233-247
    // AllocFailed 同模式：直接构造变体值断言 matches! 与 Display 非空）。
    // 实证 26-error.md L175-181：InvalidStorageMode 字段结构为
    // { operation, expected: StorageKindTag, actual: StorageKindTag, shape: Option<Vec<usize>>, conversion: Option<StorageConversionKind> }。
    let mode_err = XenonError::InvalidStorageMode {
        operation: std::borrow::Cow::Borrowed("probe"),
        expected: StorageKindTag::Owned,
        actual: StorageKindTag::Shared,
        shape: None,
        conversion: None,
    };
    assert!(matches!(mode_err, XenonError::InvalidStorageMode { .. }),
        "expected InvalidStorageMode variant existence");
    assert!(!format!("{mode_err}").is_empty(),
        "Display must be non-empty for InvalidStorageMode");
}

#[test]
fn test_layout_state_query_error_context() {
    // LayoutState 相关查询失败必须携带结构化上下文（中优先级）
    let t = Tensor2::<f64>::from_shape_vec([2, 3], vec![0.0; 6]).unwrap();
    // W20T3 仅提供 transpose()、未提供 permuted_axes()；transpose() 在 2D 上
    // 等价于完整 axis 反转，足以产生非 F-连续视图以验证 layout query error。
    // 实证 07-tensor.md §5.5 L544：`as_slice(&self) -> Option<&[A]>`——返 Option 非 Result，
    // 且未提供 `as_slice_contiguous` API。从非连续视图检查 LayoutState 选择其他路径：
    let view = t.transpose();
    // 非连续视图的 as_slice() 返 None（§5.5）——这是推荐检查点。
    assert!(view.as_slice().is_none(),
        "transposed view must be non-contiguous (as_slice should return None)");
    // 验证 layout flags 反映非连续状态（layout query 不返 Result 类型的 API 由 W6 提供）。
    // 实证 07-tensor.md L309：`pub fn flags(&self) -> LayoutFlags`（名称是 flags()不是 layout_flags()）。
    let flags = view.flags();
    assert!(!flags.is_f_contiguous(),
        "transposed view must not be F-contiguous; layout flags: {:?}", flags);
}

#[test]
fn test_workspace_error_structured_fields() {
    // 对 WorkspaceErrorCategory 存在的子变体逐一做 matches! 结构化断言。
    // 变体名严格对齐 design/26-error.md §5.3.2 L306-334：
    // AllocFailed / InvalidLayout / BorrowConflict / SplitOutOfBounds /
    // SplitCountInvariant / GrowOverflow / TypedViewRejected。

    // 1. InvalidLayout（非 2 幂对齐）—— design/24-workspace.md L246-258：
    //    Workspace::new(capacity, alignment) 在 alignment 非法时返回 InvalidLayout。
    let err = Workspace::new(64, 3).unwrap_err();
    assert!(matches!(
        err,
        XenonError::Workspace { category: WorkspaceErrorCategory::InvalidLayout { .. }, .. }
    ));

    // 2. BorrowConflict —— design/24-workspace.md L360-371、L398-409：
    //    borrow_mut() 在已持有 borrow 时返回 BorrowConflict。
    //
    // ⚠️ 借用检查实证：`Workspace::borrow_mut(&mut self)` 借走 `&mut ws`。
    // 如果在 `_g1` 持有借用期间再写 `&mut ws as *mut _`，编译器拒绝同时存在
    // 两个 `&mut self` 引用。解决方案：**raw pointer 必须在第一次 borrow_mut 之前**
    // 创建，让 `&mut ws` 仅作为构造表达式短暂存在，转为 `*mut` 后释放借用。
    // 然后通过 raw pointer 在 `_g1` 借用期间重入，触发运行期 BorrowConflict tracker。
    // 这是 24-workspace §5.5 BorrowConflict 错误路径唯一可测手段。
    let mut ws = Workspace::new(64, 8).unwrap();
    // 关键顺序：先取 raw pointer（&mut ws 立即释放），再 borrow_mut。
    let ws_ptr: *mut Workspace = &mut ws as *mut _;
    let _g1 = ws.borrow_mut().unwrap();
    // _g1 持有借用期间，通过 raw pointer + unsafe 重入 borrow_mut 触发运行期检测。
    let err = unsafe { (*ws_ptr).borrow_mut() }.unwrap_err();
    assert!(matches!(
        err,
        XenonError::Workspace { category: WorkspaceErrorCategory::BorrowConflict { .. }, .. }
    ));
    drop(_g1);

    // 3. SplitOutOfBounds —— design/24-workspace.md L492、L808-887：
    //    split_at_mut(mid) 越界时返回 SplitOutOfBounds。
    let mut ws = Workspace::new(64, 8).unwrap();
    let err = ws.split_at_mut(usize::MAX).unwrap_err();
    assert!(matches!(
        err,
        XenonError::Workspace { category: WorkspaceErrorCategory::SplitOutOfBounds { .. }, .. }
    ));

    // 4. TypedViewRejected —— design/24-workspace.md L587-700、design/26-error.md L333：
    //    向 borrow 请求 ZST/未对齐/size overflow 的 typed view 时返回 TypedViewRejected。
    //    design/24-workspace.md §5.6 L640-662 明确 `as_maybe_uninit_typed_slice::<T>(count)`
    //    在 T 为 ZST 时立即拒绝。此处用 T=() 触发 ZST 路径。
    let mut ws = Workspace::new(64, 8).unwrap();
    let mut borrow = ws.borrow_mut().unwrap();
    let err = unsafe { borrow.as_maybe_uninit_typed_slice::<()>(1) }.unwrap_err();
    assert!(matches!(
        err,
        XenonError::Workspace { category: WorkspaceErrorCategory::TypedViewRejected { .. }, .. }
    ), "expected TypedViewRejected for ZST element, got {err:?}");
    drop(borrow);

    // 5. GrowOverflow —— design/24-workspace.md L1004-1010：
    //    ensure_capacity 请求的 min_capacity 导致内部 size_of::<T>() * count 溢出时返回 GrowOverflow。
    //    使用 usize::MAX 作为 min_capacity 必然触发溢出分支。
    let mut ws = Workspace::new(16, 8).unwrap();
    let err = ws.ensure_capacity(usize::MAX).unwrap_err();
    assert!(matches!(
        err,
        XenonError::Workspace { category: WorkspaceErrorCategory::GrowOverflow { .. }, .. }
    ), "expected GrowOverflow for usize::MAX capacity, got {err:?}");

    // 6. AllocFailed —— design/24-workspace.md L268、L1038、design/26-error.md L308：
    //    底层分配器 OOM 时返回 AllocFailed { size, align }。稳定 Rust 无法可靠在
    //    单元测试内模拟 OOM（需要 feature(allocator_api) + TestingAllocator），
    //    按 28-tests §6.5 延后策略，本测试只做"变体存在性 + Display 健康"类型层级证据；
    //    真正的 OOM 触发由 W28-benchmark 的压力基线或集成 CI job 承担。
    let alloc_failed = XenonError::Workspace {
        operation: std::borrow::Cow::Borrowed("allocate"),
        category: WorkspaceErrorCategory::AllocFailed { size: 0, align: 1 },
        cause: None,
    };
    assert!(matches!(
        &alloc_failed,
        XenonError::Workspace { category: WorkspaceErrorCategory::AllocFailed { .. }, .. }
    ));
    assert!(!format!("{alloc_failed}").is_empty(),
        "Display must be non-empty for AllocFailed");

    // 7. SplitCountInvariant —— design/26-error.md L320：
    //    split 计数原子不变量被破坏（debug-only invariant check）时返回
    //    SplitCountInvariant { detail }。该变体仅在 debug 模式由内部 debug_assert + Err
    //    双重守护产生；release 不可达。本测试对其做"变体存在性 + Display 健康"验证，
    //    避免把 debug-only 路径泄漏到 release 测试矩阵（§10 平台与工程约束）。
    let split_count_inv = XenonError::Workspace {
        operation: std::borrow::Cow::Borrowed("split_at_mut"),
        category: WorkspaceErrorCategory::SplitCountInvariant {
            detail: std::borrow::Cow::Borrowed("probe"),
        },
        cause: None,
    };
    assert!(matches!(
        &split_count_inv,
        XenonError::Workspace { category: WorkspaceErrorCategory::SplitCountInvariant { .. }, .. }
    ));
    assert!(!format!("{split_count_inv}").is_empty(),
        "Display must be non-empty for SplitCountInvariant");
}

#[test]
fn test_complex_c99_layout() {
    // Complex<T> 在内存中按 C99 [re, im] 布局
    use std::mem::{align_of, size_of};
    assert_eq!(size_of::<xenon::Complex<f64>>(), 2 * size_of::<f64>());
    assert_eq!(align_of::<xenon::Complex<f64>>(), align_of::<f64>());
    let c = xenon::Complex::<f64>::new(1.0, 2.0);
    let raw: [f64; 2] = unsafe { std::mem::transmute(c) };
    assert_eq!(raw, [1.0, 2.0], "C99 layout: re first, im second");
}

## 关键设计决策

| 决策 | 依据 |
|------|------|
| 每个错误测试均做 `matches!(err, XenonError::Variant { .. })` 结构化断言 | 28-tests §5.23 / §5.18 L636 要求匹配 `WorkspaceErrorCategory` 七个结构化子变体 |
| `test_workspace_error_structured_fields` 覆盖 design 真实存在的 7 个子变体（AllocFailed / InvalidLayout / BorrowConflict / SplitOutOfBounds / SplitCountInvariant / GrowOverflow / TypedViewRejected） | design/26-error.md §5.3.2 L306-334 + design/24-workspace.md §5 |
| Display 断言只校验「非空 + 关键上下文 token」，不锁定具体文案 | 28-tests §6.5 错误信息可演进 |

## 验证方式

1. 编译检查: `cargo check --tests`
2. Lint 检查: `cargo clippy --tests -- -D warnings`
3. 测试验证: `cargo test --test test_error`
4. 手动验证: 人工审阅 Display 输出，确认 shape/axis/storage/layout 上下文足以定位失败原因。

## 注意事项

- 不得用裸 `1e-10` 容差；本文件不涉及跨路径浮点比较，全部为错误路径断言。
- 命名严格对齐 §5.23 表头，避免后续审计返工。
- `XenonError::Workspace` 仅断言公开 opaque 变体字段，不依赖私有实现细节。
